\begin{frame}{Supported Network Types}
	\begin{itemize}
		\item Directed and undirected graphs
		\item Weighted networks via properties
		\item Hierarchical / clustered graphs
		\item Large static networks ($\approx 10^6$ nodes, $\approx 5\times10^6$ edges, hardware-dependent)
	\end{itemize}
\end{frame}
