\begin{frame}{Overview}
	\begin{itemize}
		\item Open-source graph visualization and analysis framework
		\item Focus on large networks (up to $\approx 10^6$ nodes)
		\item Technical Stack:
		\begin{itemize}
			\item C++ core for performance
			\item Qt GUI for user interface
			\item Python scripting for automation
		\end{itemize}
		\item \textbf{Domains:} Social, biological, software, and communication networks
		\item \textbf{Users:} Researchers, analysts, and students
	\end{itemize}
\end{frame}
