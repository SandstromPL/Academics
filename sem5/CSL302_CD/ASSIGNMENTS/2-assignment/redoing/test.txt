// SECTION 4: CLASS WITH EXTENDS

    int baseField;
    
    void setBase(int value) {
        baseField = value;
    }
    
    int getBase() {
        return baseField;
    }
}

class DerivedClass extends BaseClass {
    int derivedField;

        derivedField = value;
    }

        return derivedField;
    }
    
    int getSum() {

}

// Recursive function
int factorial(int n) {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}
