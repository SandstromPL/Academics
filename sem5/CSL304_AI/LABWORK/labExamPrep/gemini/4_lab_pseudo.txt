// Generate all possible successor states from the current state
FUNCTION generate_successors(state):
    successors = an empty list
    FOR each heap_size in the state:
        FOR stones_to_remove from 1 to heap_size:
            create a new_state by removing stones_to_remove from the current heap
            add the new_state to the successors list
    RETURN successors

// Alpha-Beta Pruning implementation
FUNCTION alpha_beta_pruning(state, is_max_player, alpha, beta, counters, verbose, depth):

    IF the sum of stones in the state is 0:

    IF is_max_player:
        best_value = -infinity
        FOR each next_state in generate_successors(state):
            value = recursively call alpha_beta_pruning for the minimizing player (is_max_player = False)
            best_value = max(best_value, value)
            alpha = max(alpha, best_value)
            // Pruning condition
            IF alpha >= beta:
                increment the pruned nodes counter
                break from the loop
        RETURN best_value
    ELSE (minimizing player):
        best_value = +infinity
        FOR each next_state in generate_successors(state):
            value = recursively call alpha_beta_pruning for the maximizing player (is_max_player = True)
            best_value = min(best_value, value)
            beta = min(beta, best_value)
            // Pruning condition
            IF alpha >= beta:
                increment the pruned nodes counter
                break from the loop
        RETURN best_value
